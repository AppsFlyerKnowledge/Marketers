<Cards>
  <Card title="Google Play Games on PC integration with AppsFlyer" href="/docs/google-play-games-on-pc-integration-with-appsflyer" icon="fa-solid fa-mobile-screen">
    Integrate your Google Play Games on PC/Android game or app with AppsFlyer to measure the performance of campaigns marketing these apps.
  </Card>
  <Card title="Native PC integration with AppsFlyer" href="/docs/native-pc-integration-with-appsflyer" icon="fa-solid fa-desktop">

  </Card>
  <Card title="ChatGPT plugin attribution" href="/docs/chatgpt-plugin-attribution" icon="fa-solid fa-diagram-project">

  <Card title="Steam integration with AppsFlyer" href="/docs/steam-integration-with-appsflyer" icon="fa-solid fa-desktop">

  </Card>
  <Card title="Epic integration with AppsFlyer" href="/docs/epic-integration-with-appsflyer" icon="fa-solid fa-puzzle-piece">
